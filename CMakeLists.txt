cmake_minimum_required(VERSION 3.20)
project(SAB2_CrossPlatform_Foundation VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SAB2_WITH_OPENCV "Enable video/camera/ONNX pipelines" ON)
option(SAB2_BUILD_TESTS "Build portable core tests" ON)

add_library(sab2_core STATIC src/tracker.cpp src/controller.cpp)
target_include_directories(sab2_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(MSVC)
  target_compile_options(sab2_core PRIVATE /W4)
else()
  target_compile_options(sab2_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

set(PLATFORM_SOURCE src/input_stub.cpp)
if(APPLE)
  enable_language(OBJCXX)
  set(PLATFORM_SOURCE src/input_macos.mm)
elseif(WIN32)
  set(PLATFORM_SOURCE src/input_windows.cpp)
endif()
add_library(sab2_platform STATIC ${PLATFORM_SOURCE})
target_include_directories(sab2_platform PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(APPLE)
  find_library(APPLICATION_SERVICES ApplicationServices REQUIRED)
  target_link_libraries(sab2_platform PRIVATE ${APPLICATION_SERVICES})
endif()
if(WIN32)
  target_compile_definitions(sab2_platform PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()

add_executable(sab2_demo src/main.cpp)
target_link_libraries(sab2_demo PRIVATE sab2_core sab2_platform)

if(SAB2_WITH_OPENCV)
  find_package(OpenCV QUIET COMPONENTS core imgproc videoio dnn highgui)
  if(OpenCV_FOUND)
    add_library(sab2_opencv STATIC src/opencv_pipeline.cpp)
    target_include_directories(sab2_opencv PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(sab2_opencv PUBLIC ${OpenCV_LIBS} sab2_core)
    target_link_libraries(sab2_demo PRIVATE sab2_opencv)
    target_compile_definitions(sab2_demo PRIVATE SAB2_WITH_OPENCV=1)
    message(STATUS "OpenCV enabled: ${OpenCV_VERSION}")
  else()
    message(STATUS "OpenCV not installed: synthetic demo and core remain available")
  endif()
endif()

if(SAB2_BUILD_TESTS)
  enable_testing()
  add_executable(sab2_core_tests tests/core_tests.cpp)
  target_link_libraries(sab2_core_tests PRIVATE sab2_core)
  add_test(NAME sab2_core_tests COMMAND sab2_core_tests)
endif()
